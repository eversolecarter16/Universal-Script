local function SelectTab(Name)

    for PageName, Page in pairs(Pages) do

        Page.Visible =
            PageName == Name
    end

    for TabName, Button in pairs(Tabs) do

        if TabName == Name then

            Button.BackgroundColor3 =
                Config.Themes[Config.Theme]

            Button.BackgroundColor3 =
                Color3.fromRGB(30, 31, 38)

            Button.TextColor3 =
                Color3.fromRGB(155, 157, 170)
        end
    end
end

    Value.TextSize = 13
    Value.Font = Enum.Font.GothamBold

    Value.TextXAlignment =
        Enum.TextXAlignment.Right

    Value.ZIndex = 22

    local Bar =
        Instance.new("TextButton")

    Bar.Position =
        UDim2.fromOffset(16, 43)

    Bar.Size =
        UDim2.new(1, -32, 0, 8)

    Bar.BackgroundColor3 =
        Color3.fromRGB(55, 56, 66)

    Bar.Text = ""

    Bar.AutoButtonColor = false

    Bar.ZIndex = 22
    Bar.Parent = Holder

    local BarCorner =
        Instance.new("UICorner")

    BarCorner.CornerRadius =
        UDim.new(1, 0)

    BarCorner.Parent = Bar

    local Fill =
        Instance.new("Frame")

    local InitialPercent =
        math.clamp(
            (Default - Min) /
            (Max - Min),
            0,
            1
        )

    Fill.Size =
        UDim2.new(
            InitialPercent,
            0,
            1,
            0
        )

    Fill.BackgroundColor3 =
        Config.Themes[Config.Theme]

    Fill.BorderSizePixel = 0

    Fill.ZIndex = 23
    Fill.Parent = Bar

    local FillCorner =
        Instance.new("UICorner")

    FillCorner.CornerRadius =
        UDim.new(1, 0)

    FillCorner.Parent = Fill

    local SliderDragging = false

    local function SetSlider(X)

        if Bar.AbsoluteSize.X <= 0 then
            return
        end

        local Percent =
            math.clamp(
                (X - Bar.AbsolutePosition.X) /
                Bar.AbsoluteSize.X,
                0,
                1
            )

        local Number =
            math.floor(
                Min +
                ((Max - Min) * Percent)
                + 0.5
            )

        Fill.Size =
            UDim2.new(
                Percent,
                0,
                1,
                0
            )

        Value.Text =
            tostring(Number)

        Callback(Number)
    end

    Bar.MouseButton1Down:Connect(
        function()

            SliderDragging = true

            SetSlider(
                UserInputService:GetMouseLocation().X
            )
        end
    )

    UserInputService.InputChanged:Connect(
        function(Input)

            if SliderDragging
                and Input.UserInputType ==
                Enum.UserInputType.MouseMovement then

                SetSlider(
                    Input.Position.X
                )
            end
        end
    )

    UserInputService.InputEnded:Connect(
        function(Input)

            if Input.UserInputType ==
                Enum.UserInputType.MouseButton1 then

                SliderDragging = false
            end
        end
    )

    return Holder
end

local ESPObjects = {}

local function RemoveESP(TargetPlayer)

    local Highlight =
        ESPObjects[TargetPlayer]

    if Highlight then

        Highlight:Destroy()

        ESPObjects[TargetPlayer] = nil
    end
end

local function CreateESP(TargetPlayer)

    if TargetPlayer == Player then
        return
    end

    local TargetCharacter =
        TargetPlayer.Character

    if not TargetCharacter then
        return
    end

    RemoveESP(TargetPlayer)

    local Highlight =
        Instance.new("Highlight")

    Highlight.Name =
        "PlesneviyPlayerESP"

    Highlight.Adornee =
        TargetCharacter

    Highlight.DepthMode =
        Enum.HighlightDepthMode.AlwaysOnTop

    Highlight.FillColor =
        Config.Themes[Config.Theme]

    Highlight.FillTransparency = 0.65

    Highlight.OutlineColor =
        Color3.fromRGB(
            255,
            255,
            255
        )

    Highlight.OutlineTransparency = 0

    Highlight.Parent = Gui

    ESPObjects[TargetPlayer] =
        Highlight
end

local function UpdateESP()

    if not Config.PlayerESP then

        for TargetPlayer in pairs(ESPObjects) do
            RemoveESP(TargetPlayer)
        end

        return
    end

    for _, TargetPlayer in ipairs(
        Players:GetPlayers()
    ) do

        if TargetPlayer ~= Player then
            CreateESP(TargetPlayer)
        end
    end
end

--==============================================================
-- TARGET SEARCH
--==============================================================

local function GetClosestTarget()

    local ClosestPlayer = nil

    local ClosestDistance =
        Config.FOV

    local ViewportSize =
        Camera.ViewportSize

    local ScreenCenter =
        Vector2.new(
            ViewportSize.X / 2,
            ViewportSize.Y / 2
        )

    for _, TargetPlayer in ipairs(
        Players:GetPlayers()
    ) do

        if TargetPlayer ~= Player then

            local TargetCharacter =
                TargetPlayer.Character

            if TargetCharacter then

                local TargetHumanoid =
                    TargetCharacter:
                    FindFirstChildOfClass(
                        "Humanoid"
                    )

                local Head =
                    TargetCharacter:
                    FindFirstChild("Head")

                if TargetHumanoid
                    and Head
                    and TargetHumanoid.Health > 0 then

                    local Position,
                        Visible =
                        Camera:WorldToViewportPoint(
                            Head.Position
                        )

                    if Visible
                        and Position.Z > 0 then

                        local ScreenPosition =
                            Vector2.new(
                                Position.X,
                                Position.Y
                            )

                        local Distance =
                            (
                                ScreenPosition -
                                ScreenCenter
                            ).Magnitude

                        if Distance <
                            ClosestDistance then

                            ClosestDistance =
                                Distance

                            ClosestPlayer =
                                TargetPlayer
                        end
                    end
                end
            end
        end
    end

    return ClosestPlayer
end

--==============================================================
-- AIM SYSTEM
--==============================================================

local AimbotConnection

local function StartAimbot()

    if AimbotConnection then

        AimbotConnection:Disconnect()

        AimbotConnection = nil
    end

    AimbotConnection =
        RunService.RenderStepped:Connect(
            function()

                if not Config.Aimbot then
                    return
                end

                if not UserInputService:
                    IsMouseButtonPressed(
                        Enum.UserInputType.MouseButton2
                    ) then

                    return
                end

                local Target =
                    GetClosestTarget()

                if not Target then
                    return
                end

                local TargetCharacter =
                    Target.Character

                if not TargetCharacter then
                    return
                end

                local Head =
                    TargetCharacter:
                    FindFirstChild("Head")

                if not Head then
                    return
                end

                local CameraPosition =
                    Camera.CFrame.Position

                local TargetCFrame =
                    CFrame.lookAt(
                        CameraPosition,
                        Head.Position
                    )

                local Smooth =
                    math.clamp(
                        Config.AimbotSmoothness /
                        100,
                        0.01,
                        1
                    )

                -- Higher smoothness = slower,
                -- more gradual camera movement.
                local Alpha =
                    math.clamp(
                        1 - Smooth,
                        0.01,
                        0.99
                    )

                Camera.CFrame =
                    Camera.CFrame:Lerp(
                        TargetCFrame,
                        Alpha
                    )
            end
        )
end

StartAimbot()

    if Root then

        local Velocity =
            Root:FindFirstChild(
                "PlesneviyFlyVelocity"
            )

        if Velocity then
            Velocity:Destroy()
        end
    end
end

local function StartFly()

    StopFly()

    if not Root then
        return
    end

    local Velocity =
        Instance.new("BodyVelocity")

    Velocity.Name =
        "PlesneviyFlyVelocity"

    Velocity.MaxForce =
        Vector3.new(
            math.huge,
            math.huge,
            math.huge
        )

    Velocity.Velocity =
        Vector3.zero

    Velocity.Parent = Root

    FlyConnection =
        RunService.RenderStepped:Connect(
            function()

                if not Config.Fly
                    or not Root
                    or not Character then

                    return
                end

                local Direction =
                    Vector3.zero

                local CurrentCamera =
                    workspace.CurrentCamera

                if not CurrentCamera then
                    return
                end

                if UserInputService:IsKeyDown(
                    Enum.KeyCode.W
                ) then

                    Direction +=
                        CurrentCamera.CFrame.LookVector
                end

                if UserInputService:IsKeyDown(
                    Enum.KeyCode.S
                ) then

                    Direction -=
                        CurrentCamera.CFrame.LookVector
                end

                if UserInputService:IsKeyDown(
                    Enum.KeyCode.A
                ) then

                    Direction -=
                        CurrentCamera.CFrame.RightVector
                end

                if UserInputService:IsKeyDown(
                    Enum.KeyCode.D
                ) then

                    Direction +=
                        CurrentCamera.CFrame.RightVector
                end

                if UserInputService:IsKeyDown(
                    Enum.KeyCode.Space
                ) then

                    Direction +=
                        Vector3.new(
                            0,
                            1,
                            0
                        )
                end

                if UserInputService:IsKeyDown(
                    Enum.KeyCode.LeftControl
                ) then

                    Direction -=
                        Vector3.new(
                            0,
                            1,
                            0
                        )
                end

                if Direction.Magnitude > 0 then

                    Direction =
                        Direction.Unit
                end

                Velocity.Velocity =
                    Direction *
                    Config.FlySpeed
            end
        )
end

local PreviousFlyState = false

RunService.Heartbeat:Connect(
    function()

        if Config.Fly ~=
            PreviousFlyState then

            PreviousFlyState =
                Config.Fly

            if Config.Fly then
                StartFly()
            else
                StopFly()
            end
        end
    end
)

-- NPC FREEZE
--==============================================================

local FrozenNPCs = {}

local function IsNPC(Object)

    if not Object:IsA("Model") then
        return false
    end

    if Players:GetPlayerFromCharacter(
        Object
    ) then

        return false
    end

    return Object:
        FindFirstChildOfClass(
            "Humanoid"
        ) ~= nil
end

local function FreezeNPC(Model)

    local NPCHumanoid =
        Model:FindFirstChildOfClass(
            "Humanoid"
        )

    if not NPCHumanoid then
        return
    end

    if not FrozenNPCs[Model] then

        FrozenNPCs[Model] = {

            WalkSpeed =
                NPCHumanoid.WalkSpeed,

            JumpPower =
                NPCHumanoid.JumpPower,

            AutoRotate =
                NPCHumanoid.AutoRotate
        }
    end

    NPCHumanoid.WalkSpeed = 0
    NPCHumanoid.JumpPower = 0
    NPCHumanoid.AutoRotate = false
end

local function UnfreezeNPC(Model)

    local Data =
        FrozenNPCs[Model]

    local NPCHumanoid =
        Model:FindFirstChildOfClass(
            "Humanoid"
        )

    if NPCHumanoid and Data then

        NPCHumanoid.WalkSpeed =
            Data.WalkSpeed

        NPCHumanoid.JumpPower =
            Data.JumpPower

        NPCHumanoid.AutoRotate =
            Data.AutoRotate
    end

    FrozenNPCs[Model] = nil
end

RunService.Heartbeat:Connect(
    function()

        for _, Object in ipairs(
            workspace:GetChildren()
        ) do

            if IsNPC(Object) then

                if Config.FreezeNPC then

                    FreezeNPC(Object)

                elseif FrozenNPCs[Object] then

                    UnfreezeNPC(Object)
                end
            end
        end
    end
)

local MenuVisible = true
local MenuDisabled = false

UserInputService.InputBegan:Connect(
    function(Input, Processed)

        if Processed then
            return
        end

        -- RIGHT SHIFT
        if Input.KeyCode ==
            Enum.KeyCode.RightShift then

            if not MenuDisabled then

                MenuVisible =
                    not MenuVisible

                Main.Visible =
                    MenuVisible
            end
        end

        -- RIGHT ALT
        if Input.KeyCode ==
            Enum.KeyCode.RightAlt then

            if not MenuDisabled then

                MenuVisible =
                    not MenuVisible

                Main.Visible =
                    MenuVisible
            end
        end

        -- RIGHT CONTROL
        if Input.KeyCode ==
            Enum.KeyCode.RightControl then

            MenuDisabled = true
            MenuVisible = false

            Main.Visible = false
        end
    end
)
